// 큐비트 수는 20개 이상이어야 함

// CX 게이트
cx q[0], q[4];
// CY 게이트
cy q[1], q[5];
// CZ 게이트
cz q[2], q[6];

// CCX 게이트
ccx q[3], q[7], q[11];
// CCRY 게이트
cry(2.71828) q[4], q[8];
// CCZ 게이트
ccz q[5], q[9], q[13];

cswap q[10], q[14], q[18];
// CU1 게이트
cu1(3.14159) q[12], q[16];
// CU2 게이트
cu2(3.14159, 2.71828) q[15], q[19];

// 측정
measure q;
