OPENQASM 2.0;
include "qelib1.inc";

// 큐비트 수는 20개 이상이어야 함
qreg q[24];

// 초기 상태 설정 (모든 큐비트를 0으로 설정)
x q;

// U1 게이트
u1(3.14159) q[0];
// U2 게이트
u2(3.14159, 2.71828) q[1];
// U3 게이트
u3(3.14159, 2.71828, 1.61803) q[2];

// CX 게이트
cx q[0], q[4];
// CY 게이트
cy q[1], q[5];
// CZ 게이트
cz q[2], q[6];

// CCX 게이트
ccx q[3], q[7], q[11];
// CCRY 게이트
cry(2.71828) q[4], q[8];
// CCZ 게이트
ccz q[5], q[9], q[13];

// CSWAP 게이트
cswap q[10], q[14], q[18];
// CU1 게이트
cu1(3.14159) q[12], q[16];
// CU2 게이트
cu2(3.14159, 2.71828) q[15], q[19];

// 측정
measure q;

OpenQASM은 Open Quantum Assembly Language의 약자로, 양자 회로를 프로그래밍하는 언어입니다. 여기에서는 20キュ비트 이상, 3종류 이상의 양자 게이트를 전부 활용한 복잡한 양자회로를 OpenQASM 형태로 작성해 보겠습니다.
